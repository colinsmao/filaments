cmake_minimum_required(VERSION 3.22)
project(filaments LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG a74efa0d5628b74adc0426af4c5710e287fa7c2c)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 0af55ccecd98d4e5a8d1fad7de25ba429d60e863)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG f5befd2d29e66809cd1110a152e375a7f1981f06)
FetchContent_MakeAvailable(glfw glm imgui)
add_library(glad STATIC third_party/glad/src/glad.c)
target_include_directories(glad PUBLIC third_party/glad/include)
add_library(imgui_backend STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui_backend PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(imgui_backend PUBLIC glfw)
add_executable(filaments src/main.cpp)
target_link_libraries(filaments PRIVATE glad glfw glm::glm imgui_backend)
target_compile_definitions(filaments PRIVATE GLFW_INCLUDE_NONE)
if(MSVC)
    target_compile_options(filaments PRIVATE /W4)
else()
    target_compile_options(filaments PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
    target_link_options(filaments PRIVATE -static)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(settings_test tests/settings_test.cpp)
    target_include_directories(settings_test PRIVATE src)
    if(MINGW)
        target_link_options(settings_test PRIVATE -static)
    endif()
    add_test(NAME settings COMMAND settings_test "${CMAKE_CURRENT_BINARY_DIR}/settings-test/controls.ini")
    add_test(NAME renderer_smoke COMMAND filaments --smoke-test)
endif()
