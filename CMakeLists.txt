cmake_minimum_required(VERSION 3.22)
project(filaments LANGUAGES C CXX)
if(NOT WIN32)
    message(FATAL_ERROR "The Python process bridge currently targets native Windows; the Python generators are portable.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG a74efa0d5628b74adc0426af4c5710e287fa7c2c)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 0af55ccecd98d4e5a8d1fad7de25ba429d60e863)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG f5befd2d29e66809cd1110a152e375a7f1981f06)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_MakeAvailable(glfw glm imgui json)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy; print(numpy.__version__)"
    RESULT_VARIABLE NUMPY_RESULT OUTPUT_VARIABLE NUMPY_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT NUMPY_RESULT EQUAL 0)
    message(FATAL_ERROR "NumPy is required. Install python/requirements.txt with ${Python3_EXECUTABLE} -m pip install -r python/requirements.txt")
endif()
message(STATUS "Generator Python: ${Python3_EXECUTABLE}; NumPy ${NUMPY_VERSION}")
configure_file(src/runtime_config.h.in generated/runtime_config.h @ONLY)
add_library(generation STATIC src/generation.cpp)
target_include_directories(generation PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(generation PUBLIC glm::glm nlohmann_json::nlohmann_json)
add_library(glad STATIC third_party/glad/src/glad.c)
target_include_directories(glad PUBLIC third_party/glad/include)
add_library(imgui_backend STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui_backend PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(imgui_backend PUBLIC glfw)
add_executable(filaments src/main.cpp)
target_link_libraries(filaments PRIVATE glad glfw glm::glm imgui_backend generation)
target_compile_definitions(filaments PRIVATE GLFW_INCLUDE_NONE)
if(MSVC)
    target_compile_options(filaments PRIVATE /W4)
else()
    target_compile_options(filaments PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
    target_link_options(filaments PRIVATE -static)
endif()
include(CTest)
